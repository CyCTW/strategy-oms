cmake_minimum_required(VERSION 3.20)
project(oms_index_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OMS_SANITIZE "Build tests with address/undefined sanitizers" OFF)
set(OMS_SANITIZERS "address,undefined" CACHE STRING "Clang sanitizer list")
set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
add_subdirectory(third_party/abseil-cpp EXCLUDE_FROM_ALL)
add_library(oms_index INTERFACE)
target_include_directories(oms_index INTERFACE include)
target_include_directories(oms_index SYSTEM INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/abseil-cpp")
target_link_libraries(oms_index INTERFACE absl::btree)
target_compile_options(oms_index INTERFACE -Wall -Wextra -Wpedantic -Werror)
if(OMS_SANITIZE)
  target_compile_options(oms_index INTERFACE -O1 -g -fsanitize=${OMS_SANITIZERS} -fno-omit-frame-pointer)
  target_link_options(oms_index INTERFACE -fsanitize=${OMS_SANITIZERS})
endif()
add_executable(index_tests tests.cpp)
target_link_libraries(index_tests PRIVATE oms_index absl::flat_hash_map)
add_executable(clustered_bench bench.cpp)
target_link_libraries(clustered_bench PRIVATE oms_index)
add_executable(clustered_alloc bench.cpp)
target_compile_definitions(clustered_alloc PRIVATE OMS_ALLOC_COUNT=1)
target_link_libraries(clustered_alloc PRIVATE oms_index)
add_executable(linear_bench linear_bench.cpp)
target_link_libraries(linear_bench PRIVATE oms_index)
add_executable(linear_alloc linear_bench.cpp)
target_compile_definitions(linear_alloc PRIVATE OMS_ALLOC_COUNT=1)
target_link_libraries(linear_alloc PRIVATE oms_index)
add_executable(oms_parity parity/main.cpp)
target_link_libraries(oms_parity PRIVATE oms_index absl::flat_hash_map)
add_executable(price_dual_bench price_dual_bench.cpp)
target_link_libraries(price_dual_bench PRIVATE oms_index absl::flat_hash_map)
add_executable(price_dual_alloc price_dual_bench.cpp)
target_compile_definitions(price_dual_alloc PRIVATE OMS_ALLOC_COUNT=1)
target_link_libraries(price_dual_alloc PRIVATE oms_index absl::flat_hash_map)
enable_testing()
add_test(NAME index_tests COMMAND index_tests)
